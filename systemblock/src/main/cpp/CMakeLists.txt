cmake_minimum_required(VERSION 3.22)
project(adblock_vpn LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_Declare(mbedtls URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.5/mbedtls-3.6.5.tar.bz2 URL_HASH SHA256=4a11f1777bb95bf4ad96721cac945a26e04bf19f57d905f241fe77ebeddf46d8)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(fmt URL https://github.com/fmtlib/fmt/archive/refs/tags/11.2.0.tar.gz URL_HASH SHA256=bc23066d87ab3168f27cef3e97d545fa63314f5c79df5ea444d41d56f962c6af)
FetchContent_Declare(asio URL https://github.com/chriskohlhoff/asio/archive/refs/tags/asio-1-30-2.tar.gz URL_HASH SHA256=755bd7f85a4b269c67ae0ea254907c078d408cce8e1a352ad2ed664d233780e8)
FetchContent_Declare(lz4 URL https://github.com/lz4/lz4/archive/refs/tags/v1.10.0.tar.gz URL_HASH SHA256=537512904744b35e232912055ccf8ec66d768639ff3abe5788d90d792ec5f48b)
FetchContent_MakeAvailable(mbedtls fmt asio lz4)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CORE "${ROOT}/native/deps/openvpn3")
add_library(ubgo SHARED IMPORTED)
set_target_properties(ubgo PROPERTIES IMPORTED_LOCATION "${ROOT}/systemblock/build/go/${ANDROID_ABI}/libubgo.so")
add_library(ubvpn SHARED engine.cpp "${CORE}/client/ovpncli.cpp" "${CORE}/openvpn/crypto/data_epoch.cpp" "${lz4_SOURCE_DIR}/lib/lz4.c")
target_include_directories(ubvpn PRIVATE "${CORE}" "${ROOT}/native/go" "${asio_SOURCE_DIR}/asio/include" "${lz4_SOURCE_DIR}/lib" "${ROOT}/systemblock/build/go/${ANDROID_ABI}")
target_compile_definitions(ubvpn PRIVATE ASIO_STANDALONE USE_ASIO USE_MBEDTLS HAVE_LZ4)
target_compile_options(ubvpn PRIVATE -fvisibility=hidden)
target_link_options(ubvpn PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(ubvpn ubgo mbedtls mbedx509 mbedcrypto fmt::fmt log)
